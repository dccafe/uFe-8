Exemplo 1: Seq de fibonnacci

fibo:
0:	MOV		#1, R0 
2:	MOV		#2, R1 

loop:
4:	MOV		R1, R2 
5:	ADD		R0, R1 
6:	MOV		R2, R0 
7:	jmp		loop   

----------------------------------------
mult: 
00:	MOV #4, R2		(D2, 04)	// Args: R2 x R3 
02:	MOV #3, R3		(D3, 03)

04:	MOV #0, R0		(D0, 00)	// Limpa R0 e R1 
06:	MOV #0, R1		(D1, 00)

mult_loop:
08:	ADD R2, R0		(08)	    // Acumula R2 em R0, R3 vezes
09:	JC 	mult_c		(FC, 0D)	// Esses dois saltos 
0B:	JMP mult_dec	(FF, 0F)	// simulam o JNC (jump if not carry)

mult_c:
0D:	ADD #1, R1		(11, 01)	// Carry vai para MSByte

mult_dec:
0F:	ADD #-1, R3 	(13, FF)	// decrementa R3 
11:	JNZ mult_loop 	(FE, 08)	// até chegar em zero
								// * Teste de LD e ST * //
13: MOV #0x80, R3	(D3, 80)	// R3 aponta para a RAM
15: ST R0, @R3 		(A3)		// Grava LSByte
16:	ADD #1, R3		(13, 01)	// Avança o ponteiro
18: ST R1, @R3 		(A7)		// Grava o MSByte

19: ADD #0xFF, R3	(13, FF)	// Rebobina o ponteiro 
1B: LD @R3, R1		(8D)		// Carrega o LSByte em R1 

1C: XOR 0xFF, R0	(50, FF)	// Toma o complemento de 2
1E: ADD #1, R0		(10, 01)	// de R0

20: ADD R0, R1		(01)		// Subtrai R1 - R0
21: JZ  tudo_ok		(FD, 25)	// Se o resultado for 0,

erro:							// Infelizmente não funcionou
23:	JMP $			(FF, 23)	// Trava o programa aqui

tudo_ok: 						// Deu certo, 
25: JMP $			(FF, 25)	// pode parar por aqui
